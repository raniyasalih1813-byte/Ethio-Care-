class Database:
    def __init__(self, db_name=DB_NAME):
        self.db_name = db_name
        self.initialize()

    def connect(self):
        conn = sqlite3.connect(self.db_name)
        conn.row_factory = sqlite3.Row
        return conn

    def execute(self, query, params=()):
        conn = self.connect()
        try:
            cursor = conn.cursor()
            cursor.execute(query, params)
            conn.commit()
            return cursor.lastrowid
        finally:
            conn.close()

    def fetchone(self, query, params=()):
        conn = self.connect()
        try:
            cursor = conn.cursor()
            cursor.execute(query, params)
            return cursor.fetchone()
        finally:
            conn.close()

    def fetchall(self, query, params=()):
        conn = self.connect()
        try:
            cursor = conn.cursor()
            cursor.execute(query, params)
            return cursor.fetchall()
        finally:
            conn.close()

    def table_exists(self, table_name):
        row = self.fetchone(
            "SELECT name FROM sqlite_master WHERE type='table' AND name=?",
            (table_name,)
        )
        return row is not None

    def columns(self, table_name):
        if not self.table_exists(table_name):
            return []
        rows = self.fetchall(f"PRAGMA table_info({table_name})")
        return [row["name"] for row in rows]

    def initialize(self):
        self.create_users()
        self.create_administrators()
        self.create_services()
        self.create_contacts()
        self.repair_reports_table()
        self.seed_data()

    def create_users(self):
        self.execute("""
            CREATE TABLE IF NOT EXISTS users (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password TEXT NOT NULL,
                created_at TEXT NOT NULL
            )
        """)

    def create_administrators(self):
        self.execute("""
            CREATE TABLE IF NOT EXISTS administrators (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT UNIQUE NOT NULL,
                password TEXT NOT NULL,
                created_at TEXT NOT NULL
            )
        """)

    def create_services(self):
        self.execute("""
            CREATE TABLE IF NOT EXISTS emergency_services (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                category TEXT NOT NULL,
                title TEXT NOT NULL,
                description TEXT NOT NULL,
                phone TEXT NOT NULL,
                image_key TEXT,
                created_at TEXT NOT NULL
            )
        """)

    def create_contacts(self):
        self.execute("""
            CREATE TABLE IF NOT EXISTS emergency_contacts (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                category TEXT NOT NULL,
                phone TEXT NOT NULL,
                address TEXT NOT NULL
            )
        """)

    def create_reports_table(self):
        self.execute("""
            CREATE TABLE IF NOT EXISTS emergency_reports (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                username TEXT NOT NULL,
                category TEXT NOT NULL,
                title TEXT NOT NULL,
                description TEXT NOT NULL,
                phone TEXT NOT NULL,
                created_at TEXT NOT NULL
            )
        """)

    def repair_reports_table(self):
        required = {
            "id",
            "username",
            "category",
            "title",
            "description",
            "phone",
            "created_at"
        }

        if not self.table_exists("emergency_reports"):
            self.create_reports_table()
            return

        current = set(self.columns("emergency_reports"))

        if required.issubset(current):
            return

        old_table = "emergency_reports_old"

        try:
            self.execute("DROP TABLE IF EXISTS emergency_reports_old")
            self.execute(
                "ALTER TABLE emergency_reports RENAME TO emergency_reports_old"
            )

            self.create_reports_table()

            old_columns = set(self.columns(old_table))
            rows = self.fetchall(f"SELECT * FROM {old_table}")

            for row in rows:
                def get_value(names, default=""):
                    for name in names:
                        if name in old_columns:
                            value = row[name]
                            if value is not None and str(value).strip():
                                return str(value)
                    return default

                username = get_value(
                    ["username", "user", "user_name"],
                    "Unknown"
                )

                category = get_value(
                    ["category", "type"],
                    "General"
                )

                title = get_value(
                    ["title", "subject", "name"],
                    "Emergency Report"
                )

                description = get_value(
                    ["description", "message", "details"],
                    ""
                )

                phone = get_value(
                    ["phone", "phone_number"],
                    ""
                )

                created_at = get_value(
                    ["created_at", "date", "timestamp"],
                    datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                )

                self.execute("""
                    INSERT INTO emergency_reports
                    (username, category, title, description, phone, created_at)
                    VALUES (?, ?, ?, ?, ?, ?)
                """, (
                    username,
                    category,
                    title,
                    description,
                    phone,
                    created_at
                ))

        except Exception:
            try:
                self.execute("DROP TABLE IF EXISTS emergency_reports")
                self.create_reports_table()
            except Exception:
                pass

    def hash_password(self, password):
        return hashlib.sha256(
            password.encode("utf-8")
        ).hexdigest()

    def seed_data(self):
        admin = self.fetchone(
            "SELECT id FROM administrators WHERE username=?",
            ("admin",)
        )

        if not admin:
            self.execute("""
                INSERT INTO administrators
                (username, password, created_at)
                VALUES (?, ?, ?)
            """, (
                "admin",
                self.hash_password("admin1234"),
                datetime.now().strftime("%Y-%m-%d %H:%M:%S")
            ))

        service_count = self.fetchone(
            "SELECT COUNT(*) AS count FROM emergency_services"
        )

        if service_count["count"] == 0:
            for category, title, description, phone, image_key in SERVICE_DATA:
                self.execute("""
                    INSERT INTO emergency_services
                    (category, title, description, phone, image_key, created_at)
                    VALUES (?, ?, ?, ?, ?, ?)
                """, (
                    category,
                    title,
                    description,
                    phone,
                    image_key,
                    datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                ))

        contact_count = self.fetchone(
            "SELECT COUNT(*) AS count FROM emergency_contacts"
        )

        if contact_count["count"] == 0:
            for name, category, phone, address in CONTACT_DATA:
                self.execute("""
                    INSERT INTO emergency_contacts
                    (name, category, phone, address)
                    VALUES (?, ?, ?, ?)
                """, (
                    name,
                    category,
                    phone,
                    address
                ))

    def register_user(self, username, password):
        username = username.strip()

        if not username or not password:
            return False, "Username and password are required."

        if len(username) < 3:
            return False, "Username must contain at least 3 characters."

        if len(password) < 6:
            return False, "Password must contain at least 6 characters."

        if not re.match(r"^[A-Za-z0-9_.-]+$", username):
            return False, "Username contains invalid characters."

        existing = self.fetchone(
            "SELECT id FROM users WHERE username=?",
            (username,)
        )

        if existing:
            return False, "Username already exists."

        self.execute("""
            INSERT INTO users
            (username, password, created_at)
            VALUES (?, ?, ?)
        """, (
            username,
            self.hash_password(password),
            datetime.now().strftime("%Y-%m-%d %H:%M:%S")
        ))

        return True, "Registration successful."

class ScrollableFrame(tk.Frame):
    def __init__(self, parent, bg=None):
        background = bg if bg else COLORS["bg"]

        super().__init__(
            parent,
            bg=background
        )

        self.canvas = tk.Canvas(
            self,
            bg=background,
            highlightthickness=0
        )

        self.scrollbar = ttk.Scrollbar(
            self,
            orient="vertical",
            command=self.canvas.yview
        )

        self.inner = tk.Frame(
            self.canvas,
            bg=background
        )

        self.window_id = self.canvas.create_window(
            (0, 0),
            window=self.inner,
            anchor="nw"
        )

        self.canvas.configure(
            yscrollcommand=self.scrollbar.set
        )

        self.canvas.pack(
            side="left",
            fill="both",
            expand=True
        )

        self.scrollbar.pack(
            side="right",
            fill="y"
        )

        self.inner.bind(
            "<Configure>",
            self.on_inner_configure
        )

        self.canvas.bind(
            "<Configure>",
            self.on_canvas_configure
        )

        self.canvas.bind_all(
            "<MouseWheel>",
            self.mousewheel
        )

    def on_inner_configure(self, event):
        self.canvas.configure(
            scrollregion=self.canvas.bbox("all")
        )

    def on_canvas_configure(self, event):
        self.canvas.itemconfigure(
            self.window_id,
            width=event.width
        )

    def mousewheel(self, event):
        try:
            self.canvas.yview_scroll(
                int(-1 * (event.delta / 120)),
                "units"
            )
        except Exception:
            pass

    def service_emoji(self, category):
        return {
            "Fire": "🔥",
            "Medical": "🏥",
            "Police": "👮",
            "Rescue": "🚒",

            "Flood": "🌊"
        }.get(
            category,
            "🚨"
        )

    def load_service_image(
        self,
        image_key,
        label
    ):
        if not PIL_AVAILABLE:
            return

        if image_key in self.image_cache:
            try:
                label.configure(
                    image=self.image_cache[image_key],
                    text=""
                )
            except Exception:
                pass
            return

        url = IMAGE_URLS.get(
            image_key
        )

        if not url:
            return

        def download():
            try:
                data = urllib.request.urlopen(
                    url,
                    timeout=8
                ).read()

                self.image_queue.put(
                    (
                        image_key,
                        label,
                        data
                    )
                )
            except Exception:
                pass

        threading.Thread(
            target=download,
            daemon=True
        ).start()

    def process_image_queue(self):
        try:
            while True:
                image_key, label, data = (
                    self.image_queue.get_nowait()
                )

                if not PIL_AVAILABLE:
                    continue

                try:
                    image = Image.open(
                        BytesIO(data)
                    )

                    image.thumbnail(
                        (480, 190)
                    )

                    photo = ImageTk.PhotoImage(
                        image
                    )

                    self.image_cache[
                        image_key
                    ] = photo

                    if label.winfo_exists():
                        label.configure(
                            image=photo,
                            text=""
                        )

                except Exception:
                    pass

        except Empty:
            pass

        try:
            self.root.after(
                100,
                self.process_image_queue
            )
        except Exception:
            pass

    def create_tree(
        self,
        parent,
        columns,
        headings,
        widths
    ):
        tree = ttk.Treeview(
            parent,
            columns=columns,
            show="headings"
        )

        for column, heading, width in zip(
            columns,
            headings,
            widths
        ):
            tree.heading(
                column,
                text=heading
            )

            tree.column(
                column,
                width=width
            )

        tree.pack(
            fill="both",
            expand=True,
            padx=15,
            pady=15
        )

        return tree
