    def handle_exception(self, exc, val, tb):
        error_text = "".join(traceback.format_exception(exc, val, tb))

        try:
            with open(os.path.join(BASE_DIR, "system_error.log"), "a", encoding="utf-8") as f:
                f.write("\n" + "=" * 70 + "\n")
                f.write(datetime.now().strftime("%Y-%m-%d %H:%M:%S") + "\n")
                f.write(error_text)
        except Exception:
            pass

        messagebox.showerror(
            "System Error",
            "The system found an error, but the application will remain open.\n\n"
            + str(val)
        )

    def connect(self):
        con = sqlite3.connect(DB)
        con.row_factory = sqlite3.Row
        con.execute("PRAGMA foreign_keys = ON")
        return con

    def setup_database(self):

                email TEXT NOT NULL UNIQUE,

                hospital_id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL UNIQUE,

                email TEXT NOT NULL UNIQUE,
                password TEXT NOT NULL,
                status TEXT NOT NULL DEFAULT 'approved',
                created_at TEXT NOT NULL

                doctor_id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,

                password TEXT NOT NULL,

    def c(self, name):
        mode = "dark" if self.dark else "light"
        return self.colors[mode][name]

    def clear(self):
        for widget in self.winfo_children():
            widget.destroy()

        self.image_refs = []

    def make_scrollable(self):
        outer = tk.Frame(self, bg=self.c("bg"))
        outer.pack(fill="both", expand=True)

        canvas = tk.Canvas(
            outer,
            bg=self.c("bg"),
            highlightthickness=0
        )

        scrollbar = ttk.Scrollbar(
            outer,
            orient="vertical",
            command=canvas.yview
        )

        canvas.configure(yscrollcommand=scrollbar.set)

        scrollbar.pack(side="right", fill="y")
        canvas.pack(side="left", fill="both", expand=True)

        frame = tk.Frame(canvas, bg=self.c("bg"))

        window_id = canvas.create_window(
            (0, 0),
            window=frame,
            anchor="nw"
        )

        def resize(event):
            canvas.itemconfig(window_id, width=event.width)
            canvas.configure(scrollregion=canvas.bbox("all"))

        frame.bind("<Configure>", resize)
        canvas.bind("<Configure>", resize)

        def wheel(event):
            canvas.yview_scroll(int(-1 * (event.delta / 120)), "units")

        canvas.bind_all("<MouseWheel>", wheel)

        return frame

    def field(self, parent, label, password=False):
        tk.Label(
            parent,
            text=label,
            bg=self.c("card"),
            fg=self.c("text"),
            font=("Segoe UI", 10, "bold")
        ).pack(anchor="w", pady=(10, 4))

        row = tk.Frame(parent, bg=self.c("card"))
        row.pack(fill="x")

        entry = tk.Entry(
            row,
            bg=self.c("input"),
            fg=self.c("text"),
            insertbackground=self.c("text"),
            relief="solid",
            bd=1,
            font=("Segoe UI", 11)
        )

        if password:
            entry.config(show="•")

        entry.pack(side="left", fill="x", expand=True, ipady=8)

        if password:
            def toggle():
                if entry.cget("show"):
                    entry.config(show="")
                    show_button.config(text="Hide")
                else:
                    entry.config(show="•")
                    show_button.config(text="Show")

            show_button = tk.Button(
                row,
                text="Show",
                command=toggle,
                bg=self.c("button"),
                fg="white",
                border=0,
                font=("Segoe UI", 9, "bold"),
                padx=10,
                cursor="hand2"
            )
            show_button.pack(side="right", padx=(5, 0))

        return entry

    def card(self, parent, image_name, icon, title, description,
             color, command):

            parent,
            bg=self.c("card"),
            highlightbackground=self.c("border"),
            highlightthickness=1
        )

        image_area = tk.Frame(
            card,
            bg=self.c("border"),
            height=230
        )
        image_area.pack(fill="x", padx=5, pady=(5, 0))
        image_area.pack_propagate(False)

        path = os.path.join(BASE_DIR, image_name)

        if PIL_AVAILABLE and os.path.exists(path):
            try:
                img = Image.open(path)
                img.thumbnail((420, 230))
                photo = ImageTk.PhotoImage(img)

                image_label = tk.Label(
                    image_area,
                    image=photo,
                    bg=self.c("border")
                )
                image_label.pack(fill="both", expand=True)
                self.image_refs.append(photo)
            except Exception:
                tk.Label(
                    image_area,
                    text=icon,
                    bg=self.c("border"),
                    fg=self.c("text"),
                    font=("Segoe UI", 60)
                ).pack(expand=True)

            tk.Label(
                image_area,
                text=icon,
                bg=self.c("border"),
                fg=self.c("text"),
                font=("Segoe UI", 60)

            card,

    def login_page(self, role, title, login_command, register_command=None):
        self.clear()

        if role == "patient":
            self.current_view = self.patient_login
        elif role == "doctor":
            self.current_view = self.doctor_login
        elif role == "hospital":
            self.current_view = self.hospital_login
        else:
            self.current_view = self.admin_login

        page = self.make_scrollable()
        self.header(page, self.show_home)

        tk.Label(
            page,
            text=title,
            bg=self.c("bg"),
            fg=self.c("text"),
            font=("Segoe UI", 30, "bold")
        ).pack(pady=(55, 10))

        box = tk.Frame(
            page,
            bg=self.c("card"),
            highlightbackground=self.c("border"),
            highlightthickness=1
        )
        box.pack(
            fill="x",
            padx=350,
            pady=20
        )

        if role == "admin":
            user_entry = self.field(box, "Username")
        else:
            user_entry = self.field(box, "Email")

        password_entry = self.field(
            box,
            "Password",
            password=True
        )

        self.button(
            box,
            "Login",
            lambda: login_command(
                user_entry.get().strip(),
                password_entry.get()
            ),
            self.c("blue"),
            18
        ).pack(pady=(25, 10))

        if register_command:
            self.button(
                box,
                "Create Account",
                register_command,
                self.c("green"),
                18
            ).pack(pady=(5, 25))

    def do_patient_login(self, email, password):
        if not email or not password:
            messagebox.showwarning(
                "Missing Information",
                "Please enter email and password."
            )
            return

        con = self.connect()
        row = con.execute("""
            SELECT *
            FROM patients
            WHERE email = ? AND password = ?
        """, (email, hash_password(password))).fetchone()
        con.close()

        if row:
            self.current_user = dict(row)
            self.current_role = "patient"
            self.patient_dashboard()
        else:
            messagebox.showerror(
                "Login Failed",
                "Incorrect email or password."
            )

    def do_doctor_login(self, email, password):
        con = self.connect()
        row = con.execute("""
            SELECT *
            FROM doctors
            WHERE email = ? AND password = ? AND status = 'approved'
        """, (email, hash_password(password))).fetchone()
        con.close()

        if row:
            self.current_user = dict(row)
            self.current_role = "doctor"
            self.doctor_dashboard()
        else:

    def table(self, parent, columns, rows):
        frame = tk.Frame(parent, bg=self.c("card"))
        frame.pack(fill="both", expand=True, padx=20, pady=20)

        tree = ttk.Treeview(
            frame,
            columns=columns,
            show="headings"
        )

        yscroll = ttk.Scrollbar(
            frame,
            orient="vertical",
            command=tree.yview
        )

        xscroll = ttk.Scrollbar(
            frame,
            orient="horizontal",
            command=tree.xview
        )

        tree.configure(
            yscrollcommand=yscroll.set,
            xscrollcommand=xscroll.set
        )

        yscroll.pack(side="right", fill="y")
        xscroll.pack(side="bottom", fill="x")
        tree.pack(fill="both", expand=True)

        for column in columns:
            tree.heading(column, text=column)
            tree.column(column, width=150, anchor="center")

        for row in rows:
            tree.insert("", "end", values=row)

        return tree

    def book_appointment(self):
        self.clear()
        self.current_view = self.book_appointment

        )
        box.pack(fill="x", padx=250, pady=20)

        con = self.connect()

        hospitals = con.execute("""
            SELECT hospital_id, name
            FROM hospitals
            WHERE status = 'approved'
            ORDER BY name
        """).fetchall()

        hospital_values = [
            f"{r['hospital_id']} - {r['name']}"
            for r in hospitals
        ]

        tk.Label(
            box,
            text="Hospital",
            bg=self.c("card"),
            fg=self.c("text"),
            font=("Segoe UI", 10, "bold")
        ).pack(anchor="w", pady=(20, 4), padx=20)

        hospital_combo = ttk.Combobox(
            box,
            values=hospital_values,
            state="readonly",
            font=("Segoe UI", 11)
        )
        hospital_combo.pack(fill="x", padx=20, ipady=7)

        tk.Label(
            box,
            text="Doctor",
            bg=self.c("card"),
            fg=self.c("text"),
            font=("Segoe UI", 10, "bold")
        ).pack(anchor="w", pady=(15, 4), padx=20)

        doctor_combo = ttk.Combobox(
            box,
            state="readonly",
            font=("Segoe UI", 11)
        )
        doctor_combo.pack(fill="x", padx=20, ipady=7)

        def load_doctors(event=None):
            selected = hospital_combo.get()

            if not selected:
                return

            hospital_id = selected.split(" - ")[0]

            doctor_rows = con.execute("""
                SELECT doctor_id, name, specialization
                FROM doctors
                WHERE hospital_id = ?
                AND status = 'approved'
            """, (hospital_id,)).fetchall()

            doctor_combo["values"] = [
                f"{r['doctor_id']} - {r['name']} ({r['specialization']})"
                for r in doctor_rows
            ]

        hospital_combo.bind("<<ComboboxSelected>>", load_doctors)

        date_entry = self.field(box, "Date (YYYY-MM-DD)")
        time_entry = self.field(box, "Time (HH:MM)")
        reason_entry = self.field(box, "Reason")

        def save():
            selected_hospital = hospital_combo.get()
            selected_doctor = doctor_combo.get()

            if not selected_hospital or not selected_doctor:
                messagebox.showerror(
                    "Error",
                    "Please select hospital and doctor."
                )
                return

            if not date_entry.get().strip():
                messagebox.showerror("Error", "Please enter appointment date.")
                return

            if not time_entry.get().strip():
                messagebox.showerror("Error", "Please enter appointment time.")
                return

            if not reason_entry.get().strip():
                messagebox.showerror("Error", "Please enter the reason.")
                return

            hospital_id = int(
                selected_hospital.split(" - ")[0]
            )

            doctor_id = int(
                selected_doctor.split(" - ")[0]
            )

            try:
                datetime.strptime(
                    date_entry.get().strip(),
                    "%Y-%m-%d"
                )
            except ValueError:
                messagebox.showerror(
                    "Invalid Date",
                    "Use YYYY-MM-DD format."
                )
                return

            try:
                con.execute("""
                    INSERT INTO appointments
                    (patient_id, doctor_id, hospital_id,
                     appointment_date, appointment_time,
                     reason, status, created_at)
                    VALUES (?, ?, ?, ?, ?, ?, 'Pending', ?)
                """, (
                    self.current_user["patient_id"],
                    doctor_id,
                    hospital_id,
                    date_entry.get().strip(),
                    time_entry.get().strip(),
                    reason_entry.get().strip(),
                    datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                ))

                con.commit()

                messagebox.showinfo(
                    "Success",
                    "Appointment booked successfully."
                )

                self.patient_appointments()

            except sqlite3.Error as e:
                messagebox.showerror(
                    "Database Error",
                    str(e)
                )

        self.button(
            box,
            "Book Appointment",
            save,
            self.c("blue"),
            20
        ).pack(pady=30)

    def admin_requests(self):
        self.clear()
        self.current_view = self.admin_requests

        page = self.make_scrollable()
        self.header(page, self.admin_dashboard)

        self.title_label(page, "Hospital Registration Requests")

        con = self.connect()

        rows = con.execute("""
            SELECT
                request_id,
                name,
                address,
                phone,
                email,
                status,
                created_at
            FROM hospital_requests
            ORDER BY request_id DESC
        """).fetchall()

        con.close()

        tree = self.table(
            page,
            (
                "ID",
                "Hospital",
                "Address",
                "Phone",
                "Email",
                "Status",
                "Date"
            ),
            [tuple(r) for r in rows]
        )

        controls = tk.Frame(page, bg=self.c("bg"))
        controls.pack(pady=15)

        def process_request(action):
            selected = tree.selection()

            if not selected:
                messagebox.showwarning(
                    "Select Request",
                    "Please select a request."
                )
                return

            values = tree.item(
                selected[0],
                "values"
            )

            request_id = values[0]

            con = self.connect()

            request = con.execute("""
                SELECT *
                FROM hospital_requests
                WHERE request_id = ?
            """, (request_id,)).fetchone()

            if not request:
                con.close()
                return

            if action == "approve":
                try:
                    con.execute("""
                        INSERT OR IGNORE INTO hospitals
                        (name, address, phone, email, password,
                         status, created_at)
                        VALUES (?, ?, ?, ?, ?, 'approved', ?)
                    """, (
                        request["name"],
                        request["address"],
                        request["phone"],
                        request["email"],
                        request["password"],
                        datetime.now().strftime("%Y-%m-%d %H:%M:%S")
                    ))

                    con.execute("""
                        UPDATE hospital_requests
                        SET status = 'Approved'
                        WHERE request_id = ?
                    """, (request_id,))

                    con.commit()
                    con.close()

                    messagebox.showinfo(
                        "Approved",
                        "Hospital has been approved."
                    )

                    self.admin_requests()

                except sqlite3.Error as e:
                    con.close()
                    messagebox.showerror(
                        "Error",
                        str(e)
                    )

            else:
                con.execute("""
                    UPDATE hospital_requests
                    SET status = 'Rejected'
                    WHERE request_id = ?
                """, (request_id,))

                con.commit()
                con.close()

                messagebox.showinfo(
                    "Rejected",
                    "Hospital request rejected."
                )

                self.admin_requests()

        self.button(
            controls,
            "✓ Approve",
            lambda: process_request("approve"),
            self.c("green"),
            15
        ).pack(side="left", padx=5)

        self.button(
            controls,
            "✕ Reject",
            lambda: process_request("reject"),
            self.c("danger"),
            15
        ).pack(side="left", padx=5)

        self.header(
            page,
            self.current_view_back
        )

        self.title_label(page, "Our Hospitals")

        con = self.connect()

        hospitals = con.execute("""
            SELECT *
            FROM hospitals
            WHERE status = 'approved'
            ORDER BY name
        """).fetchall()

        for hospital in hospitals:
            box = tk.Frame(
                page,
                bg=self.c("card"),
                highlightbackground=self.c("border"),
                highlightthickness=1
            )
            box.pack(fill="x", padx=150, pady=10)

            tk.Label(
                box,
                text="🏥",
                bg=self.c("card"),
                fg=self.c("blue"),
                font=("Segoe UI", 38)
            ).pack(side="left", padx=25, pady=20)

            info = tk.Frame(box, bg=self.c("card"))
            info.pack(side="left", fill="x", expand=True, pady=20)

            tk.Label(
                info,
                text=hospital["name"],
                bg=self.c("card"),
                fg=self.c("text"),
                font=("Segoe UI", 17, "bold")
            ).pack(anchor="w")

            tk.Label(
                info,
                text=f"📍 {hospital['address']}    ☎ {hospital['phone']}",
                bg=self.c("card"),
                fg=self.c("muted"),
                font=("Segoe UI", 10)
            ).pack(anchor="w", pady=5)

            doctors = con.execute("""
                SELECT name, specialization
                FROM doctors
                WHERE hospital_id = ?
                AND status = 'approved'
            """, (hospital["hospital_id"],)).fetchall()

            doctor_text = "Doctors: " + (
                ", ".join(
                    f"{d['name']} ({d['specialization']})"
                    for d in doctors
                )
                if doctors
                else "No doctors registered yet"
            )

            tk.Label(
                info,
                text=doctor_text,
                bg=self.c("card"),
                fg=self.c("text"),
                font=("Segoe UI", 10),
                wraplength=800,
                justify="left"
            ).pack(anchor="w")

        con.close()

    def medicine_search(self):
        self.clear()
        self.current_view = self.medicine_search

        page = self.make_scrollable()
        self.header(page, self.current_view_back)

        self.title_label(page, "Medicine Search")

        box = tk.Frame(
            page,
            bg=self.c("card"),
            highlightbackground=self.c("border"),
            highlightthickness=1
        )
        box.pack(fill="x", padx=250, pady=20)

        search = self.field(
            box,
            "Search medicine"
        )

        results_frame = tk.Frame(
            page,
            bg=self.c("bg")
        )
        results_frame.pack(fill="x", padx=180, pady=20)

        def search_medicine():
            for w in results_frame.winfo_children():
                w.destroy()

            query = search.get().strip()

            if not query:
                messagebox.showwarning(
                    "Search",
                    "Enter a medicine name."
                )
                return

            con = self.connect()

            rows = con.execute("""
                SELECT name, category, description
                FROM medicines
                WHERE name LIKE ?
                OR category LIKE ?
                OR description LIKE ?
                ORDER BY name
            """, (
                f"%{query}%",
                f"%{query}%",
                f"%{query}%"
            )).fetchall()

            con.close()

            if not rows:
                tk.Label(
                    results_frame,
                    text="No medicine found.",
                    bg=self.c("bg"),
                    fg=self.c("muted"),
                    font=("Segoe UI", 13)
                ).pack(pady=30)
                return

            for medicine in rows:
                card = tk.Frame(
                    results_frame,
                    bg=self.c("card"),
                    highlightbackground=self.c("border"),
                    highlightthickness=1
                )
                card.pack(fill="x", pady=8)

                tk.Label(
                    card,
                    text="💊",
                    bg=self.c("card"),
                    fg=self.c("green"),
                    font=("Segoe UI", 30)
                ).pack(side="left", padx=25, pady=20)

                info = tk.Frame(card, bg=self.c("card"))
                info.pack(side="left", fill="x", expand=True, pady=20)

                tk.Label(
                    info,
                    text=medicine["name"],
                    bg=self.c("card"),
                    fg=self.c("text"),
                    font=("Segoe UI", 16, "bold")
                ).pack(anchor="w")

                tk.Label(
                    info,
                    text=medicine["category"],
                    bg=self.c("card"),
                    fg=self.c("green"),
                    font=("Segoe UI", 10, "bold")
                ).pack(anchor="w")

                tk.Label(
                    info,
                    text=medicine["description"],
                    bg=self.c("card"),
                    fg=self.c("muted"),
                    font=("Segoe UI", 10),
                    wraplength=750,
                    justify="left"
                ).pack(anchor="w", pady=5)

        self.button(
            box,
            "Search",
            search_medicine,
            self.c("blue"),
            15
        ).pack(pady=25)

        tk.Label(
            page,
            text="Example: search for “pain reliever”",
            bg=self.c("bg"),
            fg=self.c("muted"),
            font=("Segoe UI", 10)
        ).pack()

    def current_view_back(self):
        if self.current_role == "patient":
            self.patient_dashboard()
        elif self.current_role == "doctor":
            self.doctor_dashboard()
        elif self.current_role == "hospital":
            self.hospital_dashboard()
        elif self.current_role == "admin":
            self.admin_dashboard()
        else:
            self.show_home()

    def logout(self):
        self.current_user = None
        self.current_role = None
        self.show_home()
